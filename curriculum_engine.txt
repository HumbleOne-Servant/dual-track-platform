    def execute_curriculum_enrichment_pipeline(self):
        """Crawls pure_curriculum_vault to ingest and upgrade files smoothly."""
        if not os.path.exists(self.target_dir):
            print(f"📁 Notice: Directory [{self.target_dir}] empty. Pre-seeding layout tree...")
            self.seed_highest_level_raw_curriculum()

        for root, dirs, files in os.walk(self.target_dir):
            for file in files:
                if file.endswith(".json"):
                    file_path = os.path.join(root, file)
                    try:
                        with open(file_path, 'r', encoding='utf-8') as f:
                            raw_content = json.load(f)
                        
                        # Enrich document variables
                        upgraded_data = self.enrich_raw_file(file_path, raw_content)
                        
                        # Write high-tier structured output back to disk
                        with open(file_path, 'w', encoding='utf-8') as outfile:
                            json.dump(upgraded_data, outfile, indent=4)
                        self.processed_count += 1
                    except Exception as e:
                        print(f"⚠️ Skipping damaged node: {file} - {e}")

    def seed_highest_level_raw_curriculum(self):
        """Pre-seeds the 4-unit, 180-day structure into pure_curriculum_vault if missing."""
        for group, grades in GRADE_SYSTEM.items():
            for grade in grades:
                for subject in SUBJECTS:
                    for day in range(1, 181):
                        if day <= 45: unit = "unit_1_foundations"
                        elif day <= 90: unit = "unit_2_applications"
                        elif day <= 135: unit = "unit_3_mastery"
                        else: unit = "unit_4_advanced"
                        
                        path = os.path.join(self.target_dir, grade, subject, unit)
                        os.makedirs(path, exist_ok=True)
                        
                        mock_raw = {
                            "grade_prefix": grade, "subject_track": subject, "day": day, "unit_folder": unit,
                            "lesson_body": f"Raw structural text file parameters detailing core concepts for {subject.replace('_',' ')} day {day}."
                        }
                        with open(os.path.join(path, f"day_{day}.json"), 'w', encoding='utf-8') as f:
                            json.dump(mock_raw, f, indent=4)
